// Para el HTML: crea un identificador de instalación una sola vez.
function getALFREDDeviceId() {
  let id = localStorage.getItem("ALFRED_DEVICE_ID");
  if (!id) {
    id = crypto.randomUUID();
    localStorage.setItem("ALFRED_DEVICE_ID", id);
  }
  return id;
}
